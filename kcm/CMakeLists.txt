cmake_minimum_required(VERSION 3.20)
project(kcm_screenbot_screensaver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM 6.0 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Quick Qml Network)
find_package(KF6 6.0 REQUIRED COMPONENTS KCMUtils CoreAddons I18n)
kcmutils_add_qml_kcm(kcm_screenbot_screensaver)
target_sources(kcm_screenbot_screensaver PRIVATE screenbotkcm.cpp screenbotkcm.h locationsearch.cpp locationsearch.h)
target_link_libraries(kcm_screenbot_screensaver PRIVATE Qt6::Quick Qt6::Network KF6::KCMUtils KF6::CoreAddons KF6::I18n)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test Widgets)
    add_executable(locationsearch_test tests/locationsearch_test.cpp locationsearch.cpp screenbotkcm.cpp)
    target_include_directories(locationsearch_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(locationsearch_test PRIVATE Qt6::Test Qt6::Widgets Qt6::Quick Qt6::Network KF6::KCMUtils KF6::CoreAddons KF6::I18n)
    target_compile_definitions(locationsearch_test PRIVATE SCREENBOT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/..")
    add_test(NAME locationsearch COMMAND locationsearch_test)
    set_tests_properties(locationsearch PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
